rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    /* ============================================================
       ADMINS
       ------------------------------------------------------------
       Los UID de abajo son los "admins de respaldo": SIEMPRE tienen
       acceso, pase lo que pase, aunque el documento admin_config/
       admins se borre o se dañe por error. No los borres.

       Todos los demás administradores (los que crees desde la
       pestaña "Usuarios" del panel) los agrega el panel solo al
       documento admin_config/admins, en Firestore.
    ============================================================ */
    function isAdmin() {
      return request.auth != null && (
        request.auth.uid in [
          '8CBfiOxDeda0MmtTmSlXYIkfbx82',
          '9JYFhLkos8PpWdiLcowvmaw9xoo2'
        ]
        ||
        (
          exists(/databases/$(database)/documents/admin_config/admins) &&
          request.auth.uid in get(/databases/$(database)/documents/admin_config/admins).data.uids
        )
      );
    }

    // Cada cliente puede crear y leer SOLO su propio perfil
    match /usuarios/{userId} {
      allow read, update: if request.auth != null && request.auth.uid == userId;
      allow create: if request.auth != null && request.auth.uid == userId;
    }

    /* ============================================================
       VALIDACIÓN DE PEDIDOS: el precio de cada prenda se recalcula
       comparando contra el precio REAL de "productos" (talla + %
       de descuento), en vez de confiar en el navegador. Si el
       precio no coincide con el catálogo, el pedido se rechaza.
       Se revisan hasta 8 prendas por pedido (límite del lenguaje
       de reglas, que no permite "for").
    ============================================================ */
    function precioEsperadoItem(producto, talla) {
      let tieneTalla = talla != null && ('preciosTalla' in producto) && (talla in producto.preciosTalla);
      let base = tieneTalla ? producto.preciosTalla[talla] : producto.precio;
      let descuento = ('descuento' in producto) ? producto.descuento : 0;
      return descuento > 0 ? math.round(base * (1 - descuento / 100)) : base;
    }
    function itemPrecioOk(item) {
      let producto = get(/databases/$(database)/documents/productos/$(item.id)).data;
      return item.precio == precioEsperadoItem(producto, item.talla);
    }
    function itemOkEnIndice(items, i) {
      return i >= items.size() || itemPrecioOk(items[i]);
    }
    function todosLosItemsOk(items) {
      return itemOkEnIndice(items, 0) && itemOkEnIndice(items, 1) && itemOkEnIndice(items, 2) && itemOkEnIndice(items, 3)
          && itemOkEnIndice(items, 4) && itemOkEnIndice(items, 5) && itemOkEnIndice(items, 6) && itemOkEnIndice(items, 7);
    }
    function precioEnIndice(items, i) {
      return i < items.size() ? items[i].precio : 0;
    }
    function sumaPrecios(items) {
      return precioEnIndice(items, 0) + precioEnIndice(items, 1) + precioEnIndice(items, 2) + precioEnIndice(items, 3)
           + precioEnIndice(items, 4) + precioEnIndice(items, 5) + precioEnIndice(items, 6) + precioEnIndice(items, 7);
    }
    /* Texto opcional: si viene, es texto (o null) y no pasa del tope. */
    function textoOpcional(d, k, max) {
      return !(k in d) || d[k] == null || (d[k] is string && d[k].size() <= max);
    }
    /* Costo de envío: debe ser uno de los costos reales de las zonas. El panel
       guarda esa lista en config/envio (campo "costos") cada vez que se guardan
       los costos de envío. Mientras ese campo no exista, solo se exige >= 0
       (así el checkout no se rompe antes de guardar los costos una vez). */
    function envioPermitido(costo) {
      return !exists(/databases/$(database)/documents/config/envio)
        || !('costos' in get(/databases/$(database)/documents/config/envio).data)
        || costo in get(/databases/$(database)/documents/config/envio).data.costos;
    }
    function pedidoValido() {
      let d = request.resource.data;
      return d.keys().hasOnly([
          'usuario_uid','nombre_cliente','email_cliente','telefono_cliente','metodo_pago','tarjeta_final',
          'pse_banco','pse_tipo_persona','envio_departamento','envio_municipio','envio_zona','envio_barrio',
          'envio_vereda','envio_direccion','envio_edificio','envio_referencia','costo_envio','items','detalle',
          'subtotal','total','fecha','estado','estado_pago','acepta_terminos','acepta_terminos_fecha'
        ])
        && d.items is list && d.items.size() >= 1 && d.items.size() <= 8
        && todosLosItemsOk(d.items)
        && d.subtotal is number && d.subtotal == sumaPrecios(d.items)
        && d.costo_envio is number && d.costo_envio >= 0 && envioPermitido(d.costo_envio)
        && d.total is number && d.total == d.subtotal + d.costo_envio
        && d.nombre_cliente is string && d.nombre_cliente.size() >= 3 && d.nombre_cliente.size() <= 120
        && d.email_cliente is string && d.email_cliente.size() <= 160 && d.email_cliente.matches('^[^@ ]+@[^@ ]+[.][^@ ]+$')
        && d.telefono_cliente is string && d.telefono_cliente.size() >= 7 && d.telefono_cliente.size() <= 20
        && d.metodo_pago in ['contraentrega', 'pse', 'nequi', 'tarjeta']
        && d.envio_direccion is string && d.envio_direccion.size() >= 5 && d.envio_direccion.size() <= 200
        && d.envio_municipio is string && d.envio_municipio.size() >= 2 && d.envio_municipio.size() <= 80
        && textoOpcional(d, 'envio_departamento', 60) && textoOpcional(d, 'envio_zona', 40)
        && textoOpcional(d, 'envio_barrio', 100) && textoOpcional(d, 'envio_vereda', 100)
        && textoOpcional(d, 'envio_edificio', 100) && textoOpcional(d, 'envio_referencia', 400)
        && textoOpcional(d, 'pse_banco', 60) && textoOpcional(d, 'pse_tipo_persona', 20)
        // Un pedido NUNCA nace confirmado ni pagado: eso solo lo cambia un admin.
        && (!('estado' in d) || d.estado == 'pendiente')
        && (!('estado_pago' in d) || d.estado_pago in ['pendiente', 'por_verificar'])
        && (!('acepta_terminos' in d) || d.acepta_terminos == true)
        && d.detalle is string && d.detalle.size() <= 2000;
    }

    // Forma exacta del mensaje de contacto (enviarMensajeContacto en
    // js/store.js). Sin campos de más y con tamaños razonables.
    function contactoValido() {
      let d = request.resource.data;
      return d.keys().hasOnly(['nombre','correo','telefono','asunto','mensaje','fecha','origen','con_foto'])
        && d.keys().hasAll(['nombre','correo','telefono','asunto','mensaje','fecha'])
        && d.nombre is string && d.nombre.size() >= 3 && d.nombre.size() <= 80
        && d.correo is string && d.correo.size() <= 120 && d.correo.matches('^[^@ ]+@[^@ ]+[.][^@ ]+$')
        && d.telefono is string && d.telefono.size() >= 7 && d.telefono.size() <= 20
        && d.asunto is string && d.asunto.size() >= 2 && d.asunto.size() <= 60
        && d.mensaje is string && d.mensaje.size() >= 10 && d.mensaje.size() <= 1000
        && d.fecha is number
        && (!('con_foto' in d) || d.con_foto == true);
    }

    /* Lo ÚNICO que un admin puede cambiar en un mensaje de contacto ya
       recibido: su gestión (estado, leído, historial). Nombre, correo,
       teléfono, asunto, mensaje y fecha quedan intocables. */
    function gestionContactoValida() {
      let antes = resource.data;
      let despues = request.resource.data;
      return despues.diff(antes).affectedKeys().hasOnly(['estado', 'leido', 'estado_fecha', 'estado_por', 'historial_estados'])
        && (!('estado' in despues) || despues.estado in ['nuevo', 'en_revision', 'respondido'])
        && (!('leido' in despues) || despues.leido is bool)
        && (!('historial_estados' in despues) || (despues.historial_estados is list && despues.historial_estados.size() <= 200));
    }

    /* Lo ÚNICO que un admin puede cambiar en un pedido ya creado: su estado de
       gestión y su estado de pago (cada uno con su historial). Productos, precios,
       subtotal, envío, total, datos del cliente y fecha quedan intocables.
       - "estado_pago" solo admite pendiente / por_verificar / pagado y SOLO lo cambia
         un admin (el cliente nunca puede: pedidoValido lo obliga a nacer sin pagar).
       - Los historiales (historial_estados, historial_pagos) solo pueden crecer: no
         se borran ni se reescriben entradas anteriores. */
    function gestionPedidoValida() {
      let antes = resource.data;
      let despues = request.resource.data;
      return despues.diff(antes).affectedKeys().hasOnly(['estado', 'historial_estados', 'estado_pago', 'historial_pagos'])
        && ((!('estado' in antes) && !('estado' in despues))
            || despues.estado in ['pendiente', 'confirmado', 'enviado', 'entregado', 'cancelado'])
        && ((!('estado_pago' in antes) && !('estado_pago' in despues))
            || despues.estado_pago in ['pendiente', 'por_verificar', 'pagado'])
        && (!('historial_estados' in despues)
            || (despues.historial_estados is list
                && despues.historial_estados.size() <= 200
                && (!('historial_estados' in antes)
                    || antes.historial_estados.toSet().difference(despues.historial_estados.toSet()).size() == 0)))
        && (!('historial_pagos' in despues)
            || (despues.historial_pagos is list
                && despues.historial_pagos.size() <= 200
                && (!('historial_pagos' in antes)
                    || antes.historial_pagos.toSet().difference(despues.historial_pagos.toSet()).size() == 0)));
    }

    // Pedidos:
    // - "get" (leer UNO) abierto: lo usa seguimiento.html.
    // - "list" (todos) solo admin: así nadie saca la lista de clientes.
    // - "create" abierto (checkout sin cuenta) pero validado.
    // - update solo admin y SOLO estado/pago e historiales (gestionPedidoValida); delete solo admin.
    match /pedidos/{pedidoId} {
      allow get: if true;
      allow list: if isAdmin();
      // El id debe ser un id automático de Firestore (20 letras/números): el checkout
      // lo reutiliza al reintentar, y así un mismo intento nunca crea dos pedidos.
      allow create: if pedidoId.matches('[A-Za-z0-9]{20}') && pedidoValido();
      allow update: if isAdmin() && gestionPedidoValida();
      allow delete: if isAdmin();
    }

    // Descuento de stock en el checkout: el navegador del cliente (que
    // nunca es admin) baja el stock en la misma transacción que crea el
    // pedido. Solo se permite BAJAR stock y no tocar ningún otro campo.
    function soloDescuentaStock() {
      let antes = resource.data;
      let despues = request.resource.data;
      return despues.diff(antes).affectedKeys().hasOnly(['stock', 'stockColores', 'stockTallas'])
        && (!('stock' in despues) || (despues.stock is number && antes.stock is number && despues.stock >= 0 && despues.stock <= antes.stock));
    }

    // Promedio de reseñas: al publicar una reseña, el navegador del cliente
    // actualiza SOLO "rating" y "resenas" del producto. Se exige que el conteo
    // suba exactamente de 1 en 1 y que el promedio quede entre 1 y 5.
    function soloActualizaResenas() {
      let antes = resource.data;
      let despues = request.resource.data;
      let conteoAntes = ('resenas' in antes && antes.resenas is number) ? antes.resenas : 0;
      return despues.diff(antes).affectedKeys().hasOnly(['rating', 'resenas'])
        && despues.resenas is int && despues.resenas == conteoAntes + 1
        && despues.rating is number && despues.rating >= 1 && despues.rating <= 5;
    }

    // Catálogo: lectura pública. Crear/borrar y cualquier otro cambio
    // (precio, fotos, nombre...) solo admin. Bajar stock también el checkout
    // y el promedio de reseñas lo actualiza quien publica una reseña.
    match /productos/{productoId} {
      allow read: if true;
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || soloDescuentaStock() || soloActualizaResenas();
    }

    // Reseñas de productos (producto.html y sección del Inicio):
    // - lectura pública;
    // - create: cualquier visitante, solo con la forma exacta del formulario
    //   (nombre, estrellas 1-5, comentario hasta 500 letras) y para un producto que exista;
    // - update / delete: solo admin (así se borran las reseñas malas desde el panel o la consola).
    match /resenas/{resenaId} {
      allow read: if true;
      allow create: if request.resource.data.keys().hasOnly(['productoId','nombre','estrellas','comentario','fecha','con_foto'])
        && request.resource.data.keys().hasAll(['productoId','nombre','estrellas','fecha'])
        && request.resource.data.productoId is string && request.resource.data.productoId.size() <= 40
        && exists(/databases/$(database)/documents/productos/$(request.resource.data.productoId))
        && request.resource.data.nombre is string && request.resource.data.nombre.size() >= 1 && request.resource.data.nombre.size() <= 60
        && request.resource.data.estrellas is int && request.resource.data.estrellas >= 1 && request.resource.data.estrellas <= 5
        && (!('comentario' in request.resource.data) || (request.resource.data.comentario is string && request.resource.data.comentario.size() <= 500))
        && request.resource.data.fecha is string && request.resource.data.fecha.size() <= 40
        && (!('con_foto' in request.resource.data) || request.resource.data.con_foto == true);
      allow update, delete: if isAdmin();
    }

    // Foto opcional de una reseña (JPG reducido en base64, mismo id que la reseña).
    // - get abierto (la tienda la muestra); list cerrado (nadie baja todas de golpe).
    // - create: solo en el mismo lote que crea su reseña marcada con_foto.
    // - update / delete: solo admin (para quitar fotos que no deban estar).
    match /resenas_fotos/{resenaId} {
      allow get: if true;
      allow list: if false;
      allow create: if request.resource.data.keys().hasOnly(['foto','fecha'])
                    && request.resource.data.foto is string
                    && request.resource.data.foto.matches('^data:image/jpeg;base64,.*')
                    && request.resource.data.foto.size() <= 400000
                    && request.resource.data.fecha is string && request.resource.data.fecha.size() <= 40
                    && existsAfter(/databases/$(database)/documents/resenas/$(resenaId))
                    && getAfter(/databases/$(database)/documents/resenas/$(resenaId)).data.get('con_foto', false) == true;
      allow update, delete: if isAdmin();
    }

    // Config del "drop" y de los costos de envío: lectura pública,
    // escritura solo admin.
    match /config/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Borrador del drop: solo el admin lo lee o lo escribe. Lo que ve la
    // tienda es config/drop, que solo se escribe al pulsar "Publicar Drop".
    match /drop_borradores/{docId} {
      allow read, write: if isAdmin();
    }

    // Recuperación de contraseña por teléfono.
    match /telefonos/{telefono} {
      allow get: if true;
      allow list: if false;
      allow create: if request.auth != null;
      allow update, delete: if false;
    }

    // Correos del Voltage Club: cualquiera puede dejar su correo
    // (create; update si ya existía, porque el sitio guarda con
    // merge:true). Solo el admin puede leer la lista o borrar.
    match /voltage_club/{correoId} {
      allow create, update: if request.resource.data.correo is string
        && request.resource.data.correo.matches('.*@.*')
        && request.resource.data.correo.size() <= 160;
      allow read, list: if isAdmin();
      allow delete: if isAdmin();
    }

    // Mensajes del formulario de contacto (servicios.html).
    // - create: cualquier visitante, solo con la forma exacta del formulario.
    // - get / list / delete: SOLO admin.
    // - update: (a) un admin, solo para cambiar la gestión (estado, leído,
    //   historial); (b) cualquiera, solo si los datos son IDÉNTICOS
    //   (reintento del mismo envío si se cortó la conexión).
    match /mensajes_contacto/{mensajeId} {
      allow create: if contactoValido();
      allow update: if (isAdmin() && gestionContactoValida())
                    || request.resource.data == resource.data;
      allow get, list, delete: if isAdmin();
    }

    // Foto opcional del mensaje de contacto (JPG reducido en base64, mismo id del mensaje).
    // Cualquiera crea (o repite idéntica); solo el admin la lee o borra.
    match /mensajes_contacto_fotos/{mensajeId} {
      allow create: if request.resource.data.keys().hasOnly(['foto','fecha'])
                    && request.resource.data.foto is string
                    && request.resource.data.foto.matches('^data:image/jpeg;base64,.*')
                    && request.resource.data.foto.size() <= 700000
                    && request.resource.data.fecha is number;
      allow update: if request.resource.data == resource.data;
      allow get, list, delete: if isAdmin();
    }

    // Usuarios y sesiones del panel: SOLO admin.
    match /admin_usuarios/{usuarioId} {
      allow read, write: if isAdmin();
    }
    match /admin_sesiones/{usuarioId} {
      allow read, write: if isAdmin();
    }

    // Lista de UIDs admin que mantiene el panel. Solo admin.
    match /admin_config/{docId} {
      allow read, write: if isAdmin();
    }

    // Código de acceso al panel por CORREO: "get" abierto (pantalla
    // "olvidé mi código"), "list" cerrado, escritura solo admin.
    match /panel_codigos/{correoId} {
      allow get: if true;
      allow list: if false;
      allow write: if isAdmin();
    }
  }
}
