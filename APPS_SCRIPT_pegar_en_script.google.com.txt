/* ---------- 2) doPost: recibe los avisos de la tienda ---------- */
function doPost(e) {
  try {
    const datos = JSON.parse(e.postData.contents);

    if (datos.tipo === 'pedido_nuevo') {
      guardarPedidoEnSheet(datos.pedido);
      enviarCorreoConfirmacion(datos.pedido);
    }
    if (datos.tipo === 'pedido_eliminado') {
      eliminarPedidoDeSheet(datos.id);
    }
    if (datos.tipo === 'codigo_panel_recuperacion') {
      enviarCorreoCodigoPanel(datos.correo, datos.codigo);
    }
    if (datos.tipo === 'aviso_drop_nuevo') {
      enviarAvisoDropVoltageClub(datos.correos, datos.asunto, datos.mensaje);
    }
    if (datos.tipo === 'voltage_club_bienvenida') {
      enviarBienvenidaVoltageClub(datos.correo);
    }
    if (datos.tipo === 'contacto_nuevo') {
      enviarCorreoContacto(datos);
    }
    // 'problema' y 'pedido_estado' no hacen nada aquí (no rompen nada).

    return ContentService.createTextOutput('ok');
  } catch (err) {
    // La tienda usa "no-cors": el error solo se ve en "Ejecuciones".
    console.error(err);
    return ContentService.createTextOutput('error: ' + err.message);
  }
}

/* ---------- 4) HELPERS ---------- */
function nombreMetodoPago(metodo) {
  if (metodo === 'tarjeta') return 'Tarjeta';
  if (metodo === 'nequi') return 'Nequi';
  return 'Contraentrega';
}

function esPagoDeUnaVez(metodo) {
  return metodo === 'tarjeta' || metodo === 'nequi';
}

function normalizarCodigo(valor) {
  return String(valor === null || valor === undefined ? '' : valor).trim().toUpperCase();
}

function sinCerosIniciales(codigo) {
  return /^\d+$/.test(codigo) ? codigo.replace(/^0+(?=\d)/, '') : codigo;
}

/* Escapa texto para meterlo seguro en HTML. */
function esc(t) {
  return String(t === null || t === undefined ? '' : t)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
}

/* Formato de dinero: 125000 -> "125.000" (no depende del idioma del servidor). */
function dinero(n) {
  return String(Math.round(Number(n) || 0)).replace(/\B(?=(\d{3})+(?!\d))/g, '.');
}

/* Correo de la cuenta dueña del script (siempre disponible). */
function correoPropio() {
  return Session.getEffectiveUser().getEmail();
}

/* ---------- 5) PEDIDOS EN LA HOJA ---------- */
function guardarPedidoEnSheet(pedido) {
  if (!pedido) return;

  const candado = LockService.getScriptLock();
  candado.waitLock(20000);
  try {
    const hoja = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_TAB);
    if (!hoja) throw new Error('No se encontró la pestaña "' + SHEET_TAB + '"');

    const detalle = pedido.detalle ||
      (pedido.items || []).map(function (it) { return it.nombre + ' (' + it.talla + ')'; }).join(' | ');

    const direccion = [pedido.envio_direccion, pedido.envio_edificio, pedido.envio_barrio, pedido.envio_referencia]
      .filter(Boolean).join(', ');

    const fila = [
      String(pedido.codigo_pedido || pedido.id || ''),                 // A - N° Pedido (texto)
      new Date(pedido.fecha || Date.now()),                            // B - Fecha
      pedido.nombre_cliente || '',                                     // C - Cliente
      pedido.email_cliente || '',                                      // D - Correo
      pedido.telefono_cliente || '',                                   // E - Teléfono
      detalle,                                                         // F - Productos
      pedido.envio_municipio || '',                                    // G - Municipio
      direccion,                                                       // H - Dirección
      nombreMetodoPago(pedido.metodo_pago),                            // I - Método de pago
      pedido.subtotal || 0,                                            // J - Subtotal
      pedido.costo_envio || 0,                                         // K - Envío
      pedido.total || 0,                                               // L - Total
      esPagoDeUnaVez(pedido.metodo_pago) ? 'Pagado' : 'Pendiente',     // M - Estado de pago
    ];

    const numeroFila = hoja.getLastRow() + 1;
    hoja.getRange(numeroFila, 1).setNumberFormat('@'); // columna A como texto
    hoja.getRange(numeroFila, 1, 1, fila.length).setValues([fila]);
  } finally {
    candado.releaseLock();
  }
}

function eliminarPedidoDeSheet(id) {
  if (!id) { console.warn('pedido_eliminado llegó sin id.'); return; }

  const candado = LockService.getScriptLock();
  candado.waitLock(20000);
  try {
    const hoja = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_TAB);
    if (!hoja) { console.error('No se encontró la pestaña "' + SHEET_TAB + '"'); return; }

    const idCompleto = normalizarCodigo(id);
    const codigo = idCompleto.slice(-6);
    const codigoSinCeros = sinCerosIniciales(codigo);

    const ultimaFila = hoja.getLastRow();
    if (ultimaFila < 2) { console.warn('La hoja no tiene pedidos que borrar.'); return; }

    const columnaA = hoja.getRange(2, 1, ultimaFila - 1, 1).getDisplayValues();

    let borradas = 0;
    for (let i = columnaA.length - 1; i >= 0; i--) {
      const celda = normalizarCodigo(columnaA[i][0]);
      if (!celda) continue;
      if (celda === codigo || celda === idCompleto || sinCerosIniciales(celda) === codigoSinCeros) {
        hoja.deleteRow(i + 2);
        borradas++;
      }
    }

    if (borradas === 0) {
      console.warn('No se halló ninguna fila para el pedido ' + codigo + ' (id ' + idCompleto + ').');
    } else {
      console.log('Pedido ' + codigo + ': ' + borradas + ' fila(s) borrada(s) de la hoja.');
    }
  } finally {
    candado.releaseLock();
  }
}

/* Aviso de drop al Voltage Club, en tandas (por la cuota diaria de Gmail). */
function enviarAvisoDropVoltageClub(correos, asunto, mensaje) {
  if (!correos || !correos.length) return;
  if (!asunto || !mensaje) return;

  const MAX_POR_TANDA = 45;
  const propio = correoPropio();
  let enviados = 0;
  for (let i = 0; i < correos.length; i += MAX_POR_TANDA) {
    const tanda = correos.slice(i, i + MAX_POR_TANDA);
    try {
      MailApp.sendEmail({
        to: propio,
        bcc: tanda.join(','),
        subject: 'MAQUINA — ' + asunto,
        body: mensaje,
        name: 'MAQUINA',
      });
      enviados += tanda.length;
    } catch (err) {
      console.error('Tanda de aviso de drop falló (' + tanda.length + ' correos): ' + err.message);
    }
  }
  console.log('Aviso de drop: ' + enviados + ' de ' + correos.length + ' correos enviados.');
}

/* ---------- 7) DIAGNÓSTICO (se ejecuta a mano en el editor) ---------- */
function diagnosticoColumnaA() {
  const hoja = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_TAB);
  if (!hoja) { console.error('No se encontró la pestaña "' + SHEET_TAB + '"'); return; }
  const ultima = hoja.getLastRow();
  if (ultima < 2) { console.log('La hoja no tiene pedidos.'); return; }
  const valores = hoja.getRange(2, 1, ultima - 1, 1).getValues();
  let numeros = 0;
  valores.forEach(function (f, i) {
    if (typeof f[0] === 'number') {
      numeros++;
      console.log('Fila ' + (i + 2) + ': el código se guardó como NÚMERO (' + f[0] + ').');
    }
  });
  console.log('Filas revisadas: ' + valores.length + ' | códigos guardados como número: ' + numeros);
}
