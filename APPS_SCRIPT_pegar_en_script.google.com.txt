// Pega aquí el ID de tu hoja "maquina-pedidos".
// Lo sacas de la URL: https://docs.google.com/spreadsheets/d/ESTE_ES_EL_ID/edit
const SHEET_ID = "1-HvcfVP_bcXept9k0woAyAKvx9sXzjJ8Z2gYvamuC3Y";
const SHEET_TAB = "Pedidos"; // el nombre exacto de la pestaña donde están las columnas

// Tu tienda en vivo, sin barra al final. Se usa para armar el link
// de "rastrea tu pedido" que va dentro del correo de confirmación.
const TIENDA_URL = "https://maquina-super.github.io/TIENDA-MAQUINA";

/* ============================================================
   1) FUNCIÓN PRINCIPAL - recibe todos los avisos de la tienda
   ============================================================ */
function doPost(e) {
  try {
    const datos = JSON.parse(e.postData.contents);

    if (datos.tipo === 'pedido_nuevo') {
      guardarPedidoEnSheet(datos.pedido);
      enviarCorreoConfirmacion(datos.pedido);
    }

    if (datos.tipo === 'codigo_panel_recuperacion') {
      enviarCorreoCodigoPanel(datos.correo, datos.codigo);
    }

    if (datos.tipo === 'aviso_drop_nuevo') {
      enviarAvisoDropVoltageClub(datos.correos, datos.asunto, datos.mensaje);
    }

    if (datos.tipo === 'pedido_eliminado') {
      eliminarPedidoDeSheet(datos.id);
    }

    if (datos.tipo === 'voltage_club_bienvenida') {
      enviarBienvenidaVoltageClub(datos.correo);
    }

    // Este tipo lo manda la tienda pero, por ahora, este script no
    // hace nada con él (no rompe nada si llega, simplemente no
    // genera ni correo ni fila en Sheets):
    //   - datos.tipo === 'problema'   (error al guardar un pedido)
    // Si quieres que también te avise (por Telegram o correo),
    // dímelo y lo agregamos. "pedido_estado" (cambio de estado) ya
    // lo avisas por Telegram aparte, según me contaste.

    return ContentService.createTextOutput('ok');
  } catch (err) {
    // Si algo falla, igual respondemos "ok" para no romper la tienda,
    // pero queda registrado el error en "Ejecuciones" para revisarlo.
    console.error(err);
    return ContentService.createTextOutput('error: ' + err.message);
  }
}

/* ============================================================
   2) ESCRIBE LA FILA EN TU HOJA DE CÁLCULO
   OJO: revisa el orden de columnas de tu hoja y ajústalo abajo
   si no coincide (mira el arreglo "fila").
   ============================================================ */
function guardarPedidoEnSheet(pedido) {
  if (!pedido) return;
  const hoja = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_TAB);
  if (!hoja) throw new Error('No se encontró la pestaña "' + SHEET_TAB + '"');

  const detalle = pedido.detalle ||
    (pedido.items || []).map(function (it) { return it.nombre + ' (' + it.talla + ')'; }).join(' | ');

  // Dirección: todo menos el municipio, que va en su propia columna
  const direccion = [pedido.envio_direccion, pedido.envio_edificio, pedido.envio_barrio, pedido.envio_referencia]
    .filter(Boolean).join(', ');

  const fila = [
    pedido.codigo_pedido || pedido.id || '',                        // A - N° Pedido
    new Date(pedido.fecha || Date.now()),                            // B - Fecha
    pedido.nombre_cliente || '',                                     // C - Cliente
    pedido.email_cliente || '',                                      // D - Correo
    pedido.telefono_cliente || '',                                   // E - Teléfono
    detalle,                                                         // F - Productos
    pedido.envio_municipio || '',                                    // G - Municipio
    direccion,                                                       // H - Dirección
    nombreMetodoPago(pedido.metodo_pago),                            // I - Método de pago
    pedido.subtotal || 0,                                            // J - Subtotal
    pedido.costo_envio || 0,                                         // K - Envío
    pedido.total || 0,                                               // L - Total
    esPagoDeUnaVez(pedido.metodo_pago) ? 'Pagado' : 'Pendiente',      // M - Estado de pago
  ];

  hoja.appendRow(fila);
}

/* Nombre bonito del método de pago para la hoja y el correo. */
function nombreMetodoPago(metodo) {
  if (metodo === 'tarjeta') return 'Tarjeta';
  if (metodo === 'nequi') return 'Nequi';
  return 'Contraentrega';
}

/* true si el pago ya se hizo (tarjeta o Nequi), false si queda
   pendiente para cuando llegue el pedido (contraentrega). */
function esPagoDeUnaVez(metodo) {
  return metodo === 'tarjeta' || metodo === 'nequi';
}

/* ============================================================
   3) BORRA LA FILA DE LA HOJA CUANDO SE ELIMINA UN PEDIDO
   La tienda solo manda el "id" (el de Firestore). El código que
   aparece en la columna A de la hoja son los últimos 6 caracteres
   de ese id en mayúsculas (así arma "codigo_pedido" la tienda), así
   que buscamos la fila por ese mismo valor.
   ============================================================ */
function eliminarPedidoDeSheet(id) {
  if (!id) return;
  const hoja = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_TAB);
  if (!hoja) return;

  const codigo = String(id).slice(-6).toUpperCase();
  const datos = hoja.getDataRange().getValues();

  // De abajo hacia arriba, para que borrar una fila no desordene
  // los índices de las que todavía faltan por revisar.
  for (let i = datos.length - 1; i >= 1; i--) {
    const codigoFila = String(datos[i][0] || '').toUpperCase();
    if (codigoFila === codigo) {
      hoja.deleteRow(i + 1); // +1 porque las filas de Sheets empiezan en 1, no en 0
    }
  }
}

/* ============================================================
   4) CORREO DE CONFIRMACIÓN DE PEDIDO AL CLIENTE
   Si pagó de una vez (tarjeta o Nequi) dice "pago confirmado".
   Si es contraentrega, dice "pedido confirmado, pago pendiente".

    const foto = it.foto
      ? '<img src="' + it.foto + '" width="64" height="64" style="border-radius:10px;object-fit:cover;border:1px solid #ddd;" alt="' + it.nombre + '">'
      : '';
    return '<tr>' +

      '<td style="padding:8px 0 8px 12px;font-family:Arial,sans-serif;font-size:14px;color:#151512;">' +
        '<strong>' + it.nombre + '</strong><br>' +
        '<span style="color:#666;">Talla ' + it.talla + ' — $' + Number(it.precio).toLocaleString('es-CO') + '</span>' +
      '</td>' +
    '</tr>';

    .filter(Boolean).join(', ');

  // Link para que el cliente rastree su pedido cuando quiera (le
  // precarga el número, solo le falta poner su teléfono ahí mismo).

  const lineaEstadoTexto = yaPago
    ? 'Tu pago por ' + metodoTexto + ' se confirmó correctamente. Ya recibimos tu pedido #' + codigo + ' en MAQUINA.'
    : 'Ya recibimos tu pedido #' + codigo + ' en MAQUINA. Queda PENDIENTE de pago: pagas $' +

    'Subtotal: $' + Number(pedido.subtotal || 0).toLocaleString('es-CO') + '\n' +
    'Envío: $' + Number(pedido.costo_envio || 0).toLocaleString('es-CO') + '\n' +
    'Total: $' + Number(pedido.total || 0).toLocaleString('es-CO') + '\n' +
    'Estado del pago: ' + (yaPago ? 'Pagado (' + metodoTexto + ')' : 'Pendiente (contraentrega)') + '\n\n' +
    'Dirección de entrega: ' + direccion + '\n' +
    'Método de pago: ' + metodoTexto + '\n\n' +
    'Rastrea tu pedido cuando quieras: ' + linkSeguimiento + '\n\n' +
    '¿Dudas sobre tu pedido? Escríbenos por WhatsApp: https://wa.me/573239523623\n\n' +
    'Gracias por tu compra ⚡\n— MAQUINA';

  const cuerpoHtml =
    '<div style="font-family:Arial,sans-serif;max-width:480px;margin:0 auto;">' +
      '<div style="background:#b4e614;color:#151512;padding:16px 20px;border-radius:14px 14px 0 0;text-align:center;">' +
        '<div style="font-size:13px;letter-spacing:1px;text-transform:uppercase;font-weight:700;">' + encabezado + '</div>' +
      '</div>' +
      '<div style="border:1px solid #eee;border-top:none;border-radius:0 0 14px 14px;padding:20px;">' +
        '<p style="font-size:15px;color:#151512;">¡Hola <strong>' + (pedido.nombre_cliente || '') + '</strong>! ⚡</p>' +

      'entrar al panel solo con este código: también hace falta\n' +
      'tu contraseña de siempre).\n\n' +
      '— MAQUINA',
    name: 'MAQUINA',
  });
}

/* ============================================================
   6) AVISO DE DROP NUEVO AL VOLTAGE CLUB
   ============================================================ */
function enviarAvisoDropVoltageClub(correos, asunto, mensaje) {
  if (!correos || !correos.length) return;
  if (!asunto || !mensaje) return;

  MailApp.sendEmail({
    to: Session.getActiveUser().getEmail(),
    bcc: correos.join(','),
    subject: 'MAQUINA — ' + asunto,
    body: mensaje,
    name: 'MAQUINA',
  });
}
